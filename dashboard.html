-- Tifa Inventory — Restrict Add/Delete to Admins Only -- Run this in Supabase → SQL Editor → New Query -- Employees currently have full read/write/insert/delete within their -- assigned categories, via the "employees scoped access to items" policy -- from the original migration. This replaces that one policy with two -- narrower ones: employees keep the ability to read and update items -- (so they can still update stock counts day to day), but lose the -- ability to insert new items or delete existing ones. drop policy if exists "employees scoped access to items" on inventory_items; create policy "employees read scoped items" on inventory_items for select using ( exists ( select 1 from category_access ca where ca.user_id = auth.uid() and ca.category = inventory_items.category ) ); create policy "employees update scoped items" on inventory_items for update using ( exists ( select 1 from category_access ca where ca.user_id = auth.uid() and ca.category = inventory_items.category ) ) with check ( exists ( select 1 from category_access ca where ca.user_id = auth.uid() and ca.category = inventory_items.category ) ); -- No insert or delete policy is created for employees here. Combined with -- how Postgres RLS works (deny by default unless some policy explicitly -- allows it), that means only the existing "admins full access to items" -- policy can satisfy an insert or delete — so only admins can add or -- remove items, while employees keep full read/update access within -- whatever categories they've been granted.
